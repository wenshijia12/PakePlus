        // ================================================================
        //  滚屏控制
        // ================================================================
        let scrollTimer = null;
        let isScrolling = true;
        let scrollPaused = false;
        let isUserScrolling = false;
        let userScrollTimeout = null;

        function getRowHeight() {
            const firstRow = document.querySelector('.rank-table tbody tr:not(.hidden-row)');
            if (!firstRow) return 0;
            return firstRow.offsetHeight;
        }

        function getHeaderHeight() {
            const header = document.querySelector('.rank-table thead');
            if (!header) return 0;
            return header.offsetHeight;
        }

        function getVisibleRows() {
            const viewportHeight = scrollWrapper.clientHeight;
            const h = getHeaderHeight();
            const rh = getRowHeight();
            if (rh === 0) return 0;
            return Math.floor((viewportHeight - h) / rh);
        }

        function getTotalRows() {
            return document.querySelectorAll('.rank-table tbody tr:not(.hidden-row)').length;
        }

        function getMaxScrollTop() {
            const totalRows = getTotalRows();
            const visibleRows = getVisibleRows();
            if (totalRows <= visibleRows) return 0;
            const rh = getRowHeight();
            const headerH = getHeaderHeight();
            const contentHeight = headerH + totalRows * rh;
            const viewportHeight = scrollWrapper.clientHeight;
            return Math.max(0, contentHeight - viewportHeight);
        }

        function scrollToRow(index) {
            const rh = getRowHeight();
            const headerH = getHeaderHeight();
            const totalRows = getTotalRows();
            const visibleRows = getVisibleRows();
            if (rh === 0 || totalRows === 0) return;
            const maxIndex = Math.max(0, totalRows - visibleRows);
            const safeIndex = Math.min(Math.max(0, index), maxIndex);
            const scrollY = headerH + safeIndex * rh;
            scrollWrapper.scrollTop = Math.min(scrollY, getMaxScrollTop());
        }

        function startScroll() {
            stopScroll();
            const totalRows = getTotalRows();
            const visibleRows = getVisibleRows();
            if (totalRows <= visibleRows || totalRows <= 1) { isScrolling = false;
                updateScrollBtnState(); return; }
            isScrolling = true;
            updateScrollBtnState();
            const maxScroll = getMaxScrollTop();
            if (scrollWrapper.scrollTop >= maxScroll - 10) scrollWrapper.scrollTop = 0;
            scrollTimer = setInterval(() => {
                if (scrollPaused || isUserScrolling) return;
                const totalNow = getTotalRows();
                const visibleNow = getVisibleRows();
                if (totalNow <= visibleNow || totalNow <= 1) { stopScroll(); return; }
                const maxScroll = getMaxScrollTop();
                const currentScroll = scrollWrapper.scrollTop;
                const rh = getRowHeight();
                const headerH = getHeaderHeight();
                if (rh === 0) return;
                const rawIndex = (currentScroll - headerH) / rh;
                let nextIndex = Math.floor(rawIndex) + 1;
                const maxIndex = Math.max(0, totalNow - visibleNow);
                if (nextIndex > maxIndex) nextIndex = 0;
                const targetScroll = headerH + nextIndex * rh;
                scrollWrapper.scrollTo({ top: Math.min(targetScroll, maxScroll), behavior: 'smooth' });
            }, 3500);
        }

        function stopScroll() {
            if (scrollTimer) { clearInterval(scrollTimer);
                scrollTimer = null; }
            isScrolling = false;
            updateScrollBtnState();
        }

        function toggleScroll() {
            if (isScrolling) { stopScroll(); } else { startScroll(); }
        }

        function updateScrollBtnState() {
            if (isScrolling) {
                btnScrollToggle.classList.add('active');
                btnScrollToggle.innerHTML = '<span class="dot"></span> 滚屏';
            } else {
                btnScrollToggle.classList.remove('active');
                btnScrollToggle.innerHTML = '⏸ 暂停';
            }
        }

        scrollWrapper.addEventListener('scroll', () => {
            if (!isUserScrolling) { isUserScrolling = true; if (isScrolling && scrollTimer) { clearInterval(scrollTimer);
                    scrollTimer = null; } }
            clearTimeout(userScrollTimeout);
            userScrollTimeout = setTimeout(() => { isUserScrolling = false; if (isScrolling) startScroll(); }, 3000);
        });
        scrollWrapper.addEventListener('mouseenter', () => { scrollPaused = true; });
        scrollWrapper.addEventListener('mouseleave', () => { scrollPaused = false; });
        btnScrollToggle.addEventListener('click', toggleScroll);

        let searchKeyword = '';

        function filterPlayers(keyword) {
            searchKeyword = keyword.trim().toLowerCase();
            const rows = document.querySelectorAll('.rank-table tbody tr');
            let matchCount = 0;
            rows.forEach(row => {
                if (row.querySelector('.empty-state')) { row.classList.add('hidden-row'); return; }
                const idCell = row.querySelector('td:nth-child(3)');
                const nameCell = row.querySelector('td:nth-child(4)');
                const idText = idCell ? idCell.textContent.trim() : '';
                const nameText = nameCell ? nameCell.textContent.trim() : '';
                let matched = true;
                if (searchKeyword) {
                    matched = idText.toLowerCase().includes(searchKeyword) ||
                        nameText.toLowerCase().includes(searchKeyword);
                }
                if (matched) { row.classList.remove('hidden-row');
                    matchCount++; } else { row.classList.add('hidden-row'); }
            });
            matchCountEl.textContent = matchCount;
            if (searchKeyword && matchCount === 0) {
                searchNoResult.classList.add('visible');
                searchCount.style.display = 'none';
            } else {
                searchNoResult.classList.remove('visible');
                searchCount.style.display = '';
            }
            if (searchKeyword) { searchClear.classList.add('visible'); } else { searchClear.classList.remove('visible'); }
            if (searchKeyword) {
                if (isScrolling && scrollTimer) { clearInterval(scrollTimer);
                    scrollTimer = null; }
                const firstMatch = document.querySelector('.rank-table tbody tr:not(.hidden-row)');
                if (firstMatch) {
                    const index = Array.from(document.querySelectorAll('.rank-table tbody tr:not(.hidden-row)')).indexOf(
                        firstMatch);
                    if (index >= 0) setTimeout(() => scrollToRow(index), 100);
                }
            } else {
                if (isScrolling) startScroll();
                scrollWrapper.scrollTo({ top: 0, behavior: 'smooth' });
            }
        }

            };

        function saveData() { localStorage.setItem(STORAGE_KEY, JSON.stringify(players)); }

        function generateId() {
            const ts = Date.now().toString().slice(-6);
            const rand = Math.floor(Math.random() * 100).toString().padStart(2, '0');
            return 'P' + ts + rand;
        }

        function isIdUnique(id, excludeIndex) {
            return !players.some((p, idx) => p.id === id && idx !== excludeIndex);
        }

        function cleanId(id) { return id.trim(); }

        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        function updateStatsDisplay() {
            const total = players.length;
            totalEl.textContent = total;
            if (total === 0) { maxEl.textContent = '—'; } else { const scores = players.map(p => p.score);
                maxEl.textContent = Math.max(...scores); }
        }

        // ================================================================
        //  CORE: sort & render
        // ================================================================
        function sortPlayers() {
            const currentRankMap = {};
            players.forEach((p, idx) => { currentRankMap[p.id] = idx + 1; });
            players.sort((a, b) => b.score - a.score);
            players.forEach((p, idx) => {
                const newRank = idx + 1;
                const oldRank = previousRankings[p.id] || newRank;
                if (oldRank < newRank) p._trend = 'down';
                else if (oldRank > newRank) p._trend = 'up';
                else p._trend = 'same';
            });
            players.forEach(p => { previousRankings[p.id] = players.indexOf(p) + 1; });
        }

        function render() {
            sortPlayers();
            const total = players.length;
            updateStatsDisplay();

            if (total === 0) {
                tbody.innerHTML =
                    `<tr><td colspan="6"><div class="empty-state"><div class="empty-icon">🃏</div><p>酒馆空无一人 · 快来录入第一位勇士</p></div></td></tr>`;
                stopScroll();
                matchCountEl.textContent = '0';
                searchNoResult.classList.remove('visible');
                searchCount.style.display = '';
                return;
            }

            let html = '';
            const medals = ['🥇', '🥈', '🥉'];
            for (let i = 0; i < players.length; i++) {
                const p = players[i];
                const rank = i + 1;
                let rankDisplay = rank;
                let rowClass = '';
                if (rank === 1) { rankDisplay = medals[0];
                    rowClass = 'rank-1'; } else if (rank === 2) { rankDisplay = medals[1];
                    rowClass = 'rank-2'; } else if (rank === 3) { rankDisplay = medals[2];
                    rowClass = 'rank-3'; }

                let trendHtml = '';
                if (p._trend === 'up') trendHtml = '<span class="trend up">▲</span>';
                else if (p._trend === 'down') trendHtml = '<span class="trend down">▼</span>';
                else trendHtml = '<span class="trend same">—</span>';

                html += `
                    <tr class="${rowClass}" data-index="${i}">
                        <td><span class="medal">${rankDisplay}</span></td>
                        <td>${trendHtml}</td>
                        <td>${escapeHtml(p.id || '')}</td>
                        <td>${escapeHtml(p.name)}</td>
                        <td>${p.score}</td>
                        <td>
                            <div class="action-group">
                                <div class="quick-score">
                                    <button class="btn-q plus" data-index="${i}" data-delta="10">+10</button>
                                    <button class="btn-q plus" data-index="${i}" data-delta="50">+50</button>
                                    <button class="btn-q minus" data-index="${i}" data-delta="-10">-10</button>
                                    <button class="btn-q minus" data-index="${i}" data-delta="-50">-50</button>
                                </div>
                                <button class="btn btn-edit" data-index="${i}" data-action="edit">✎</button>
                                <button class="btn btn-delete" data-index="${i}" data-action="delete">✕</button>
                            </div>
                        </td>
                    </tr>
                `;
            }
            tbody.innerHTML = html;

            if (searchKeyword) filterPlayers(searchKeyword);
            else {
                document.querySelectorAll('.rank-table tbody tr').forEach(row => row.classList.remove('hidden-row'));
                matchCountEl.textContent = players.length;
                searchNoResult.classList.remove('visible');
                searchCount.style.display = '';
                searchClear.classList.remove('visible');
            }

            document.querySelectorAll('.btn-q').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    e.stopPropagation();
                    const index = parseInt(btn.dataset.index);
                    const delta = parseInt(btn.dataset.delta);
                    if (isNaN(index) || isNaN(delta)) return;
                    if (index < 0 || index >= players.length) return;
                    const newScore = Math.max(0, players[index].score + delta);
                    players[index].score = newScore;
                    saveData();
                    render();
                });
            });

            setTimeout(() => {
                const totalNow = getTotalRows();
                const visibleNow = getVisibleRows();
                if (totalNow > visibleNow && totalNow > 1 && isScrolling && !searchKeyword) startScroll();
                else if (totalNow <= visibleNow || totalNow <= 1) stopScroll();
                else if (searchKeyword) { if (scrollTimer) { clearInterval(scrollTimer);
                        scrollTimer = null; } }
            }, 100);
        }

        function addPlayer(id, name, score) {
            const trimmedId = cleanId(id);
            const trimmedName = name.trim();
            if (!trimmedName) { alert('请输入玩家姓名'); return false; }
            if (isNaN(score) || score < 0) { alert('请输入有效的积分（≥ 0）'); return false; }
            let finalId = trimmedId;
            if (!finalId) { finalId = generateId(); } else {
                if (!isIdUnique(finalId, -1)) { alert(`ID "${finalId}" 已被其他玩家使用，请换一个。`); return false; }
                if (!/^[a-zA-Z0-9_-]+$/.test(finalId)) { alert('ID只能包含字母、数字、下划线和短横线'); return false; }
            }
            const dup = players.find(p => p.name.toLowerCase() === trimmedName.toLowerCase());
            if (dup) { if (!confirm(`已有玩家 "${dup.name}" (ID: ${dup.id})，是否继续添加？`)) return false; }
            const newPlayer = { id: finalId, name: trimmedName, score: Math.round(score) };
            players.push(newPlayer);
            previousRankings[finalId] = players.length;
            saveData();
            render();
            return true;
        }

        function deletePlayer(index) {
            if (index < 0 || index >= players.length) return;
            const name = players[index].name;
            if (!confirm(`确定要删除玩家 "${name}" 吗？`)) return;
            const id = players[index].id;
            delete previousRankings[id];
            players.splice(index, 1);
            saveData();
            render();
        }

        function editPlayer(index, newId, newName, newScore) {
            if (index < 0 || index >= players.length) return false;
            const trimmedId = cleanId(newId);
            const trimmedName = newName.trim();
            if (!trimmedName) { alert('请输入玩家姓名'); return false; }
            if (isNaN(newScore) || newScore < 0) { alert('请输入有效的积分（≥ 0）'); return false; }
            if (!trimmedId) { alert('ID不能为空'); return false; }
            if (!/^[a-zA-Z0-9_-]+$/.test(trimmedId)) { alert('ID只能包含字母、数字、下划线和短横线'); return false; }
            const oldId = players[index].id;
            if (trimmedId !== oldId && !isIdUnique(trimmedId, index)) {
                alert(`ID "${trimmedId}" 已被其他玩家使用，请换一个。`);
                return false;
            }
            if (trimmedId !== oldId) {
                const rank = previousRankings[oldId];
                delete previousRankings[oldId];
                previousRankings[trimmedId] = rank || index + 1;
            }
            players[index].id = trimmedId;
            players[index].name = trimmedName;
            players[index].score = Math.round(newScore);
            saveData();
            render();
            return true;
        }

        function openModal(index) {
            if (index < 0 || index >= players.length) return;
            editingIndex = index;

            modalOverlay.classList.remove('active');
            editingIndex = null;
            scrollPaused = false;
        }

        function handleModalSave() {
            if (editingIndex === null || editingIndex < 0 || editingIndex >= players.length) return;
            const idVal = modalId.value.trim();
            const nameVal = modalName.value.trim();
            const scoreVal = parseFloat(modalScore.value);
            if (!idVal) { alert('ID不能为空');
                modalId.focus(); return; }
            if (!nameVal) { alert('请输入玩家姓名');
                modalName.focus(); return; }
            if (isNaN(scoreVal) || scoreVal < 0) { alert('请输入有效的积分（≥ 0）');
                modalScore.focus(); return; }
            const success = editPlayer(editingIndex, idVal, nameVal, scoreVal);
            if (success) closeModal();
        }
